-- Проверка типов столбцов
SELECT
    column_name,
    data_type,
    column_default,
    is_nullable
FROM
    information_schema.columns
WHERE
    table_name = 'papers_metadata'
ORDER BY
    ordinal_position;

Права для агента
CREATE ROLE agent_readonly WITH LOGIN PASSWORD '...';
GRANT CONNECT ON DATABASE micro_services TO agent_readonly;
GRANT USAGE ON SCHEMA public TO agent_readonly;
GRANT SELECT ON papers_metadata, users, chats, reviews TO agent_readonly;
ALTER DEFAULT PRIVILEGES IN SCHEMA public GRANT SELECT ON TABLES TO agent_readonly;

GRANT INSERT, UPDATE ON reviews TO agent_readonly;
GRANT USAGE, SELECT ON SEQUENCE reviews_id_seq TO agent_readonly;
GRANT UPDATE (review_flag) ON papers_metadata TO agent_readonly;

-- 2. Права на последовательность для knowledge.id (нужно для INSERT)
GRANT USAGE, SELECT ON SEQUENCE knowledge_id_seq TO agent_readonly;
